// lib/conditions.ts
import type { Condition, FormValues } from "@/types/operation";
export function evaluateCondition(cond: Condition | undefined, values: FormValues): boolean {
  if (!cond) return true;
  if ("and" in cond) return cond.and.every((c) => evaluateCondition(c, values));
  if ("or" in cond) return cond.or.some((c) => evaluateCondition(c, values));
  if ("not" in cond) return !evaluateCondition(cond.not, values);
  const v = values[cond.param];
  switch (cond.operator) {
    case "eq": return v === cond.value;
    case "neq": return v !== cond.value;
    case "in": return Array.isArray(cond.value) && cond.value.includes(v as never);
    case "notIn": return Array.isArray(cond.value) && !cond.value.includes(v as never);
    case "gt": return typeof v === "number" && v > (cond.value as number);
    case "gte": return typeof v === "number" && v >= (cond.value as number);
    case "lt": return typeof v === "number" && v < (cond.value as number);
    case "lte": return typeof v === "number" && v <= (cond.value as number);
    case "truthy": return Boolean(v);
    case "falsy": return !v;
    default: return true;
  }
}

// lib/defaults.ts
import type { OperationDescriptor, FormValues } from "@/types/operation";
export function defaultValues(op: OperationDescriptor): FormValues {
  const out: FormValues = {};
  for (const p of op.params) {
    if (p.default !== undefined) { out[p.key] = p.default; continue; }
    switch (p.type) {
      case "string": out[p.key] = ""; break;
      case "number": out[p.key] = p.min ?? 0; break;
      case "integer": out[p.key] = p.min ?? 0; break;
      case "boolean": out[p.key] = false; break;
      case "enum": out[p.key] = p.options[0]?.value ?? ""; break;
      case "multiselect": out[p.key] = []; break;
      case "color": out[p.key] = "#000000"; break;
      case "keyvalue": out[p.key] = {}; break;
      case "range": out[p.key] = { min: p.min ?? 0, max: p.max ?? 100 }; break;
      case "subparams": out[p.key] = []; break;
      default: out[p.key] = ""; break;
    }
  }
  return out;
}

// components/OperationMenu.tsx
"use client";
import { OPERATIONS, TIER_LABELS } from "@/lib/registry";
import type { OperationDescriptor } from "@/types/operation";
export function OperationMenu({ onSelect }: { onSelect: (op: OperationDescriptor) => void }) {
  const tiers = Object.keys(TIER_LABELS).map(Number).sort((a, b) => a - b);
  return (
    <nav className="space-y-6">
      {tiers.map((tier) => {
        const ops = OPERATIONS.filter((o) => o.tier === tier);
        if (!ops.length) return null;
        return (
          <section key={tier}>
            <h3 className="text-sm font-semibold uppercase tracking-wide text-gray-500">{TIER_LABELS[tier]}</h3>
            <ul className="mt-2 space-y-1">
              {ops.map((op) => (
                <li key={op.id}>
                  <button onClick={() => onSelect(op)} className="w-full rounded px-3 py-2 text-left hover:bg-gray-100">
                    <div className="font-medium">{op.name}</div>
                    <div className="text-xs text-gray-500">{op.description}</div>
                  </button>
                </li>
              ))}
            </ul>
          </section>
        );
      })}
    </nav>
  );
}

// components/OperationForm.tsx
"use client";
import { useMemo, useState } from "react";
import type { OperationDescriptor, ParamDef, FormValues } from "@/types/operation";
import { defaultValues } from "@/lib/defaults";
import { evaluateCondition } from "@/lib/conditions";
import { ParamInput } from "./ParamInput";
export function OperationForm({ op }: { op: OperationDescriptor }) {
  const [values, setValues] = useState<FormValues>(() => defaultValues(op));
  const visibleParams = useMemo(() => op.params.filter((p) => evaluateCondition(p.showIf, values)), [op.params, values]);
  const groups = useMemo(() => {
    const map = new Map<string, ParamDef[]>();
    for (const p of visibleParams) {
      const g = p.group ?? "General";
      if (!map.has(g)) map.set(g, []);
      map.get(g)!.push(p);
    }
    return map;
  }, [visibleParams]);
  const setValue = (key: string, v: unknown) => setValues((prev) => ({ ...prev, [key]: v }));
  return (
    <form className="space-y-6">
      <header>
        <h2 className="text-lg font-semibold">{op.name}</h2>
        <p className="text-sm text-gray-600">{op.description}</p>
      </header>
      {[...groups.entries()].map(([group, params]) => (
        <fieldset key={group} className="space-y-3">
          <legend className="text-sm font-medium text-gray-700">{group}</legend>
          {params.map((p) => (
            <ParamInput key={p.key} param={p} value={values[p.key]} disabled={evaluateCondition(p.disableIf, values)} onChange={(v) => setValue(p.key, v)} />
          ))}
        </fieldset>
      ))}
      <pre className="rounded bg-gray-50 p-3 text-xs">{JSON.stringify(values, null, 2)}</pre>
    </form>
  );
}

// components/ParamInput.tsx
"use client";
import type { ParamDef } from "@/types/operation";
type Props = { param: ParamDef; value: unknown; disabled?: boolean; onChange: (v: unknown) => void };
export function ParamInput({ param, value, disabled, onChange }: Props) {
  const common = "w-full rounded border px-2 py-1 text-sm disabled:opacity-50";
  switch (param.type) {
    case "string":
      return (
        <label className="block">
          <span className="text-sm">{param.label}</span>
          <input className={common} placeholder={param.placeholder} value={(value as string) ?? ""} disabled={disabled} onChange={(e) => onChange(e.target.value)} />
          {param.helpText && <p className="text-xs text-gray-500">{param.helpText}</p>}
        </label>
      );
    case "number":
    case "integer":
      return (
        <label className="block">
          <span className="text-sm">{param.label}</span>
          <input type="number" className={common} min={param.min} max={param.max} step={param.type === "integer" ? 1 : param.step ?? "any"} value={(value as number) ?? 0} disabled={disabled} onChange={(e) => onChange(Number(e.target.value))} />
          {param.unit && <span className="text-xs text-gray-500">{param.unit}</span>}
        </label>
      );
    case "boolean":
      return (
        <label className="flex items-center gap-2">
          <input type="checkbox" checked={Boolean(value)} disabled={disabled} onChange={(e) => onChange(e.target.checked)} />
          <span className="text-sm">{param.label}</span>
        </label>
      );
    case "enum":
      return (
        <label className="block">
          <span className="text-sm">{param.label}</span>
          <select className={common} value={(value as string) ?? ""} disabled={disabled} onChange={(e) => onChange(e.target.value)}>
            {param.options.map((o) => (<option key={o.value} value={o.value} disabled={o.disabled}>{o.label}</option>))}
          </select>
        </label>
      );
    case "multiselect":
      return (
        <div>
          <span className="text-sm">{param.label}</span>
          <div className="mt-1 space-y-1">
            {param.options.map((o) => {
              const arr = (value as string[]) ?? [];
              const checked = arr.includes(o.value);
              return (
                <label key={o.value} className="flex items-center gap-2">
                  <input type="checkbox" checked={checked} disabled={disabled} onChange={(e) => { const next = e.target.checked ? [...arr, o.value] : arr.filter((v) => v !== o.value); onChange(next); }} />
                  <span className="text-sm">{o.label}</span>
                </label>
              );
            })}
          </div>
        </div>
      );
    case "time":
    case "duration":
      return (
        <label className="block">
          <span className="text-sm">{param.label}</span>
          <input className={common} placeholder={param.format === "timecode" ? "00:00:00.000" : "seconds"} value={(value as string) ?? ""} disabled={disabled} onChange={(e) => onChange(e.target.value)} />
          {param.helpText && <p className="text-xs text-gray-500">{param.helpText}</p>}
        </label>
      );
    case "color":
      return (
        <label className="block">
          <span className="text-sm">{param.label}</span>
          <input type="color" className="h-8 w-full" value={typeof value === "string" ? value.slice(0, 7) : "#000000"} disabled={disabled} onChange={(e) => onChange(e.target.value)} />
        </label>
      );
    case "size":
      return (
        <label className="block">
          <span className="text-sm">{param.label}</span>
          <input className={common} value={(value as string) ?? ""} disabled={disabled} onChange={(e) => onChange(e.target.value)} />
          {param.presets && (
            <div className="mt-1 flex flex-wrap gap-1">
              {param.presets.map((p) => (<button key={p} type="button" className="rounded bg-gray-100 px-2 py-0.5 text-xs" onClick={() => onChange(p)}>{p}</button>))}
            </div>
          )}
        </label>
      );
    case "bitrate":
      return (
        <label className="block">
          <span className="text-sm">{param.label}</span>
          <input className={common} placeholder="192k" value={(value as string) ?? ""} disabled={disabled} onChange={(e) => onChange(e.target.value)} />
        </label>
      );
    case "expression":
    case "filtergraph":
      return (
        <label className="block">
          <span className="text-sm">{param.label}</span>
          <textarea className={common} rows={3} value={(value as string) ?? ""} disabled={disabled} onChange={(e) => onChange(e.target.value)} />
          {param.type === "expression" && param.variables && (<p className="text-xs text-gray-500">Variables: {param.variables.map((v) => v.name).join(", ")}</p>)}
        </label>
      );
    case "keyvalue": {
      const obj = (value as Record<string, string>) ?? {};
      return (
        <div>
          <span className="text-sm">{param.label}</span>
          <pre className="mt-1 rounded bg-gray-50 p-2 text-xs">{JSON.stringify(obj, null, 2)}</pre>
          <p className="text-xs text-gray-500">(key/value editor placeholder)</p>
        </div>
      );
    }
    case "range": {
      const r = (value as { min: number; max: number }) ?? { min: 0, max: 100 };
      return (
        <div>
          <span className="text-sm">{param.label}</span>
          <div className="flex gap-2">
            <input type="number" className={common} value={r.min} min={param.min} max={param.max} disabled={disabled} onChange={(e) => onChange({ ...r, min: Number(e.target.value) })} />
            <input type="number" className={common} value={r.max} min={param.min} max={param.max} disabled={disabled} onChange={(e) => onChange({ ...r, max: Number(e.target.value) })} />
          </div>
        </div>
      );
    }
    case "file":
      return (
        <label className="block">
          <span className="text-sm">{param.label}</span>
          <input type="file" className={common} multiple={param.multiple} disabled={disabled} onChange={(e) => onChange(param.multiple ? Array.from(e.target.files ?? []) : e.target.files?.[0] ?? null)} />
        </label>
      );
    case "codec":
    case "stream":
      return (
        <label className="block">
          <span className="text-sm">{param.label}</span>
          <input className={common} placeholder={param.type === "codec" ? "libx264" : "0:v:0"} value={(value as string) ?? ""} disabled={disabled} onChange={(e) => onChange(e.target.value)} />
          <p className="text-xs text-gray-500">(picker placeholder)</p>
        </label>
      );
    case "subparams": {
      const list = (value as Record<string, unknown>[]) ?? [];
      return (
        <div>
          <span className="text-sm">{param.label}</span>
          <button type="button" className="mt-1 rounded bg-gray-100 px-2 py-1 text-xs" onClick={() => onChange([...list, {}])}>+ Add entry</button>
          <p className="mt-1 text-xs text-gray-500">{list.length} entr{list.length === 1 ? "y" : "ies"}</p>
        </div>
      );
    }
    default: return null;
  }
}
